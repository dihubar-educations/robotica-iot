¿Cómo hace un pin para dar “medio rojo”?

El pin solo puede estar en 0 V o en 3,3 V. Lo que cambia es cuánto tiempo pasa en alto en cada ciclo. Mové el valor y bajá la frecuencia para ver qué pasa.

OSCILOSCOPIO · CH1 = pin D5 · 1 V/divbase de tiempo
CICLO DE TRABAJO
50,2 %
TIEMPO EN ALTO
502 µs
PERÍODO
1000 µs
TENSIÓN PROMEDIO
1,66 V
MULTÍMETRO · V DCD5 – GND
1,66 V
LED ROJOse ve fijo
Ciclo de trabajo = valor / 255  ·  Tiempo en alto = ciclo × período  ·  Tensión promedio = ciclo × 3,3 V

Para discutir: a 1 Hz el LED parpadea y el multímetro no se queda quieto. A 1 kHz el ojo y el multímetro “promedian” y ven un valor intermedio. ¿Desde qué frecuencia deja de notarse el parpadeo?

En la placa real, analogWrite() trabaja de 100 Hz a 60 kHz. Para probar 1, 5 y 50 Hz, el programa de abajo genera el PWM a mano.

Esquema de conexionado
  • D5 → resistencia de 150 Ω (marrón, verde, marrón) → pata R del LED
  • Pata común (−) → riel − → GND
  • Potenciómetro (opcional): extremos a 3V3 y GND, cursor a A0
  • Las patas G y B quedan sin conectar
Programa para el IDE de Arduino · s1_pwm_frecuencia.ino
Valor y frecuencia desde el Monitor serie (o con un potenciómetro en A0). Por debajo de 100 Hz hace el PWM a mano, porque analogWrite() del ESP8266 no baja de 100 Hz.
/*
  ==========================================================================
  SIMULADOR 1 - ¿Cómo hace un pin para dar "medio rojo"?
  PWM por hardware (analogWrite) y PWM hecho a mano para frecuencias bajas
  NodeMCU V3 (ESP8266) - Placa en el IDE: "NodeMCU 1.0 (ESP-12E Module)"
  ==========================================================================

  CONEXIONADO
    D5 (GPIO14) -> resistencia de 150 ohm -> pata R del LED -> GND (cátodo común)
    Potenciómetro (opcional): extremos a 3V3 y GND, cursor a A0

  CÓMO SE USA (Monitor serie a 115200 baudios, "Nueva línea")
    v128   -> valor en el pin (0 a 255)
    f5     -> frecuencia en Hz (1 a 40000)
    p      -> activa/desactiva el potenciómetro en A0 para fijar el valor
    ?      -> muestra la ayuda

  POR QUÉ HAY DOS FORMAS DE HACER PWM
    analogWrite() en el ESP8266 acepta de 100 Hz a 60 kHz. Por debajo de
    100 Hz el programa hace el PWM "a mano": mira el reloj (micros) y prende
    o apaga el pin. Así se puede ver el parpadeo a 1 Hz o 5 Hz.
  ==========================================================================
*/

const uint8_t  PIN_LED   = D5;
const float    VCC       = 3.3;
const uint32_t FREC_MIN_HARDWARE = 100;   // límite inferior de analogWriteFreq()

uint8_t  valor      = 128;     // 0 a 255
uint32_t frecuencia = 1000;    // Hz
bool     usarPot    = false;
bool     pwmPorSoftware = false;
uint32_t inicioCiclo = 0;
uint32_t ultimaLecturaPot = 0;
uint32_t ultimoInforme = 0;

void informar() {
  float ciclo    = valor / 255.0;
  float periodo  = 1000000.0 / frecuencia;   // microsegundos
  float enAlto   = ciclo * periodo;
  float promedio = ciclo * VCC;

  Serial.println(F("------------------------------------------------"));
  Serial.printf("Valor en el pin : %u / 255\n", valor);
  Serial.printf("Frecuencia      : %lu Hz (%s)\n", (unsigned long)frecuencia,
                pwmPorSoftware ? "PWM hecho a mano" : "analogWrite por hardware");
  Serial.printf("Ciclo de trabajo: %.1f %%\n", ciclo * 100.0);
  if (periodo >= 1000.0) {
    Serial.printf("Período         : %.1f ms\n", periodo / 1000.0);
    Serial.printf("Tiempo en alto  : %.1f ms\n", enAlto / 1000.0);
  } else {
    Serial.printf("Período         : %.0f us\n", periodo);
    Serial.printf("Tiempo en alto  : %.0f us\n", enAlto);
  }
  Serial.printf("Tensión promedio: %.2f V  (medila con el multímetro)\n", promedio);
}

void aplicar() {
  pwmPorSoftware = (frecuencia < FREC_MIN_HARDWARE);

  if (pwmPorSoftware) {
    analogWrite(PIN_LED, 0);        // detiene el PWM por hardware
    pinMode(PIN_LED, OUTPUT);
    inicioCiclo = micros();
  } else {
    analogWriteFreq(frecuencia);
    analogWrite(PIN_LED, valor);
  }
}

// PWM hecho a mano: dentro de cada período, el pin está en alto
// mientras el tiempo transcurrido sea menor que el tiempo en alto.
void pwmSoftware() {
  uint32_t periodo = 1000000UL / frecuencia;
  uint32_t enAlto  = (uint32_t)((uint64_t)periodo * valor / 255);
  uint32_t t       = (micros() - inicioCiclo) % periodo;
  digitalWrite(PIN_LED, (t < enAlto) ? HIGH : LOW);
}

void mostrarAyuda() {
  Serial.println();
  Serial.println(F("Comandos: v0..v255 (valor), f1..f40000 (Hz), p (potenciómetro), ? (ayuda)"));
  Serial.println(F("Probá: f1, después f5, f50, f1000. ¿Desde cuándo deja de verse el parpadeo?"));
}

void leerComando() {
  if (!Serial.available()) return;
  String linea = Serial.readStringUntil('\n');
  linea.trim();
  if (linea.length() == 0) return;

  char comando = linea.charAt(0);
  long numero  = linea.substring(1).toInt();

  switch (comando) {
    case 'v':
    case 'V':
      if (numero < 0)   numero = 0;
      if (numero > 255) numero = 255;
      valor = (uint8_t)numero;
      usarPot = false;
      aplicar();
      informar();
      break;

    case 'f':
    case 'F':
      if (numero < 1)     numero = 1;
      if (numero > 40000) numero = 40000;
      frecuencia = (uint32_t)numero;
      aplicar();
      informar();
      break;

    case 'p':
    case 'P':
      usarPot = !usarPot;
      Serial.printf("Potenciómetro %s\n", usarPot ? "ACTIVADO (A0)" : "desactivado");
      break;

    default:
      mostrarAyuda();
      break;
  }
}

void leerPotenciometro() {
  if (!usarPot || millis() - ultimaLecturaPot < 50) return;
  ultimaLecturaPot = millis();

  uint8_t nuevo = (uint8_t)(analogRead(A0) >> 2);   // 0..1023 -> 0..255
  if (abs((int)nuevo - (int)valor) >= 2) {
    valor = nuevo;
    if (!pwmPorSoftware) analogWrite(PIN_LED, valor);
    if (millis() - ultimoInforme > 400) {            // no saturar el monitor
      ultimoInforme = millis();
      informar();
    }
  }
}

void setup() {
  Serial.begin(115200);
  delay(300);
  Serial.println();
  Serial.println(F("=== Simulador 1: PWM y tensión promedio ==="));

  analogWriteRange(255);
  pinMode(PIN_LED, OUTPUT);
  aplicar();
  informar();
  mostrarAyuda();
}

void loop() {
  leerComando();
  leerPotenciometro();
  if (pwmPorSoftware) {
    pwmSoftware();
  }
}

Mezcla de colores: tres PWM a la vez

El LED RGB son tres LED en una sola cápsula. Cada uno recibe su propio PWM y el ojo suma las tres luces.

Rojo
Verde
Azul
LED RGB#FF8C00
CanalValorCicloTensión prom.

Desafío: antes de mover los deslizadores, que cada grupo prediga los tres valores para lograr amarillo, cian y naranja. Después lo comprueban.

Esquema de conexionado
  • D5 → 150 Ω → pata R
  • D6 → 100 Ω (marrón, negro, marrón) → pata G
  • D7 → 100 Ω → pata B
  • Pata común (−) → riel − → GND
Programa para el IDE de Arduino · s2_mezcla_rgb.ino
Colores por el Monitor serie como 255,140,0 o #FF8C00, y un arcoíris automático con a.
/*
  ==========================================================================
  SIMULADOR 2 - Mezcla de colores: tres PWM a la vez
  NodeMCU V3 (ESP8266) - Placa en el IDE: "NodeMCU 1.0 (ESP-12E Module)"
  ==========================================================================

  CONEXIONADO (LED RGB de cátodo común)
    D5 (GPIO14) -> 150 ohm      -> pata R
    D6 (GPIO12) -> 47 a 100 ohm -> pata G
    D7 (GPIO13) -> 47 a 100 ohm -> pata B
    Pata común  -> GND

  CÓMO SE USA (Monitor serie a 115200 baudios, "Nueva línea")
    255,140,0  -> fija rojo, verde y azul (0 a 255 cada uno)
    #FF8C00    -> el mismo color en hexadecimal, como en la web
    a          -> arcoíris automático (se detiene con cualquier color)
    ?          -> ayuda

  DESAFÍO: antes de escribir, predecí los valores para amarillo, cian y
  naranja. Después comprobalo.
  ==========================================================================
*/

const uint8_t PIN_R = D5;
const uint8_t PIN_G = D6;
const uint8_t PIN_B = D7;
const float   VCC   = 3.3;

uint8_t r = 255, g = 140, b = 0;
bool    arcoiris = false;
uint16_t tono = 0;                 // 0 a 359 grados
uint32_t ultimoPaso = 0;

void escribirLED() {
  analogWrite(PIN_R, r);
  analogWrite(PIN_G, g);
  analogWrite(PIN_B, b);
}

void filaTabla(const char* nombre, uint8_t v) {
  float ciclo = v / 255.0;
  Serial.printf("  %-6s %3u   %5.1f %%   %4.2f V\n", nombre, v, ciclo * 100.0, ciclo * VCC);
}

void informar() {
  Serial.println(F("------------------------------------------------"));
  Serial.printf("Color #%02X%02X%02X\n", r, g, b);
  Serial.println(F("  Canal  Valor  Ciclo     Tensión prom."));
  filaTabla("Rojo",  r);
  filaTabla("Verde", g);
  filaTabla("Azul",  b);
}

// Convierte un tono (0-359) con saturación y brillo máximos a RGB 0-255
void tonoARgb(uint16_t h, uint8_t &rr, uint8_t &gg, uint8_t &bb) {
  uint8_t sector = h / 60;              // 0 a 5
  uint8_t sube   = (uint8_t)((h % 60) * 255 / 60);
  uint8_t baja   = 255 - sube;
  switch (sector) {
    case 0:  rr = 255;  gg = sube; bb = 0;    break;
    case 1:  rr = baja; gg = 255;  bb = 0;    break;
    case 2:  rr = 0;    gg = 255;  bb = sube; break;
    case 3:  rr = 0;    gg = baja; bb = 255;  break;
    case 4:  rr = sube; gg = 0;    bb = 255;  break;
    default: rr = 255;  gg = 0;    bb = baja; break;
  }
}

uint8_t limitar(long v) {
  if (v < 0)   return 0;
  if (v > 255) return 255;
  return (uint8_t)v;
}

void mostrarAyuda() {
  Serial.println();
  Serial.println(F("Escribí: 255,140,0  o  #FF8C00  |  a = arcoíris  |  ? = ayuda"));
}

void leerComando() {
  if (!Serial.available()) return;
  String linea = Serial.readStringUntil('\n');
  linea.trim();
  if (linea.length() == 0) return;

  if (linea == "a" || linea == "A") {
    arcoiris = true;
    Serial.println(F("Arcoíris activado. Escribí un color para detenerlo."));
    return;
  }

  if (linea.charAt(0) == '#' && linea.length() == 7) {
    long valorHex = strtol(linea.substring(1).c_str(), nullptr, 16);
    r = (valorHex >> 16) & 0xFF;
    g = (valorHex >> 8) & 0xFF;
    b = valorHex & 0xFF;
    arcoiris = false;
    escribirLED();
    informar();
    return;
  }

  int coma1 = linea.indexOf(',');
  int coma2 = linea.indexOf(',', coma1 + 1);
  if (coma1 > 0 && coma2 > coma1) {
    r = limitar(linea.substring(0, coma1).toInt());
    g = limitar(linea.substring(coma1 + 1, coma2).toInt());
    b = limitar(linea.substring(coma2 + 1).toInt());
    arcoiris = false;
    escribirLED();
    informar();
    return;
  }

  mostrarAyuda();
}

void setup() {
  Serial.begin(115200);
  delay(300);
  Serial.println();
  Serial.println(F("=== Simulador 2: mezcla de colores RGB ==="));

  analogWriteRange(255);
  analogWriteFreq(1000);
  pinMode(PIN_R, OUTPUT);
  pinMode(PIN_G, OUTPUT);
  pinMode(PIN_B, OUTPUT);

  escribirLED();
  informar();
  mostrarAyuda();
}

void loop() {
  leerComando();

  if (arcoiris && millis() - ultimoPaso >= 20) {
    ultimoPaso = millis();
    tono = (tono + 1) % 360;
    tonoARgb(tono, r, g, b);
    escribirLED();
    if (tono % 60 == 0) informar();      // informa en cada color "puro"
  }
}

El desafío del ánodo común

Mismo programa, otro LED, colores invertidos. ¿Por qué? Elegí un color, cambiá el tipo de LED y después activá la corrección en el código.

Pediste
El LED muestraCorrecto
CanalPedidoValor en el pinLED encendido
Con ánodo común el LED enciende cuando el pin está en BAJO → valor en el pin = 255 − valor pedido
Esquema de conexionado
  • D5 → 150 Ω → pata R · D6 → 100 Ω → pata G · D7 → 100 Ω → pata B
  • Pata común (−) → riel − → GND
  • El esquema cambia cuando elegís el tipo de LED arriba.
Programa para el IDE de Arduino · s3_anodo_comun.ino
Recorre una secuencia de colores y muestra el valor pedido y el valor en el pin. Se corrige cambiando ANODO_COMUN.
/*
  ==========================================================================
  SIMULADOR 3 - El desafío del ánodo común
  NodeMCU V3 (ESP8266) - Placa en el IDE: "NodeMCU 1.0 (ESP-12E Module)"
  ==========================================================================

  CONEXIONADO
    D5 (GPIO14) -> 150 ohm      -> pata R
    D6 (GPIO12) -> 47 a 100 ohm -> pata G
    D7 (GPIO13) -> 47 a 100 ohm -> pata B
    Pata común:
      - Cátodo común -> GND
      - Ánodo común  -> 3V3  (¡NUNCA a VIN ni a 5 V!)

  QUÉ HACE
    Recorre una secuencia de colores cada 2 segundos y muestra en el
    Monitor serie (115200) el valor pedido y el valor que va al pin.

  EL DESAFÍO
    1. Cargalo con ANODO_COMUN = false y un LED de cátodo común: todo bien.
    2. Cambiá el LED por uno de ánodo común SIN tocar el código.
       ¿Qué ves cuando la secuencia pide "Apagado"? ¿Y "Rojo"?
    3. Corregí el código: ANODO_COMUN = true. ¿Qué cambió en la tabla?
  ==========================================================================
*/

const bool ANODO_COMUN = false;   // false = cátodo común | true = ánodo común

const uint8_t PIN_R = D5;
const uint8_t PIN_G = D6;
const uint8_t PIN_B = D7;

struct Color {
  const char* nombre;
  uint8_t r, g, b;
};

const Color SECUENCIA[] = {
  {"Rojo",          255,   0,   0},
  {"Verde",           0, 255,   0},
  {"Azul",            0,   0, 255},
  {"Amarillo",      255, 255,   0},
  {"Violeta tenue",  64,   0, 128},
  {"Blanco",        255, 255, 255},
  {"Apagado",         0,   0,   0},
};
const uint8_t CANTIDAD = sizeof(SECUENCIA) / sizeof(SECUENCIA[0]);

uint8_t indice = 0;
uint32_t ultimoCambio = 0;

// Cátodo común: el LED enciende con el pin en ALTO  -> se escribe igual.
// Ánodo común:  el LED enciende con el pin en BAJO  -> hay que invertir.
uint8_t valorParaPin(uint8_t pedido) {
  if (ANODO_COMUN) {
    return 255 - pedido;
  }
  return pedido;
}

void mostrarColor(const Color &c) {
  uint8_t pr = valorParaPin(c.r);
  uint8_t pg = valorParaPin(c.g);
  uint8_t pb = valorParaPin(c.b);

  analogWrite(PIN_R, pr);
  analogWrite(PIN_G, pg);
  analogWrite(PIN_B, pb);

  Serial.println(F("------------------------------------------------"));
  Serial.printf("Color pedido: %s   (código para LED de %s común)\n",
                c.nombre, ANODO_COMUN ? "ÁNODO" : "CÁTODO");
  Serial.println(F("  Canal  Pedido  Valor en el pin"));
  Serial.printf("  Rojo    %3u       %3u\n", c.r, pr);
  Serial.printf("  Verde   %3u       %3u\n", c.g, pg);
  Serial.printf("  Azul    %3u       %3u\n", c.b, pb);
}

void setup() {
  Serial.begin(115200);
  delay(300);
  Serial.println();
  Serial.println(F("=== Simulador 3: ánodo común vs cátodo común ==="));

  analogWriteRange(255);
  analogWriteFreq(1000);
  pinMode(PIN_R, OUTPUT);
  pinMode(PIN_G, OUTPUT);
  pinMode(PIN_B, OUTPUT);

  mostrarColor(SECUENCIA[0]);
  ultimoCambio = millis();
}

void loop() {
  if (millis() - ultimoCambio >= 2000) {
    ultimoCambio = millis();
    indice = (indice + 1) % CANTIDAD;
    mostrarColor(SECUENCIA[indice]);
  }
}

¿Qué resistencia pongo?

La salida del NodeMCU es de 3,3 V. Cada color de LED “se come” una tensión distinta, y lo que sobra queda en la resistencia. Conviene no pasar de unos 12 mA por pin.

I = (3,3 V − 2,0 V) / 150 Ω = 8,7 mA
8,7 mA
Bien

Escala de 0 a 20 mA. La línea marca el límite recomendado de 12 mA.

Esquema de conexionado
  • D5 → resistencia elegida → pata R del LED
  • Punto entre la resistencia y el LED (columna 10) → A0
  • Pata común (−) → riel − → GND
  • Las bandas de la resistencia cambian con el valor que elegís arriba.
Programa para el IDE de Arduino · s4_medidor_led.ino
Mide con A0 la tensión del LED y calcula la corriente real. Se calibra con el multímetro.
/*
  ==========================================================================
  SIMULADOR 4 - ¿Qué resistencia pongo? Medidor de corriente del LED
  NodeMCU V3 (ESP8266) - Placa en el IDE: "NodeMCU 1.0 (ESP-12E Module)"
  ==========================================================================

  IDEA
    El pin D5 queda fijo en ALTO. El punto entre la resistencia y el LED
    se conecta a A0: así la placa mide la tensión del LED (Vf) y calcula
    la corriente con la ley de Ohm:

        I = (V_pin - V_led) / R

  CONEXIONADO (un solo color del LED, cátodo común)
    D5 -> resistencia R -> punto medio -> pata del LED -> GND
                               |
                               +-> A0

  ANTES DE EMPEZAR: CALIBRAR CON EL MULTÍMETRO
    - V_PIN: medí la tensión de D5 contra GND con el LED conectado
      y escribila en V_PIN (suele dar un poco menos de 3,3 V).
    - ESCALA_A0: en el NodeMCU, A0 lee de 0 a ~3,3 V. Si la tensión del
      LED que muestra el programa no coincide con el multímetro, ajustá
      este valor.

  CÓMO SE USA (Monitor serie a 115200 baudios, "Nueva línea")
    r150   -> indica el valor de la resistencia que pusiste (en ohm)
    ?      -> ayuda
  ==========================================================================
*/

const uint8_t PIN_LED = D5;

float V_PIN     = 3.30;     // tensión real del pin en alto (medila)
float ESCALA_A0 = 3.30;     // tensión que corresponde a la lectura 1023
float R_OHM     = 150.0;    // resistencia colocada
const float LIMITE_MA = 12.0;

uint32_t ultimoInforme = 0;

float leerTensionA0() {
  // Promedio de 32 lecturas para reducir el ruido
  uint32_t suma = 0;
  for (uint8_t i = 0; i < 32; i++) {
    suma += analogRead(A0);
    delay(2);
  }
  float lectura = suma / 32.0;
  return lectura * ESCALA_A0 / 1023.0;
}

void mostrarAyuda() {
  Serial.println(F("Comandos: r<ohm> (por ejemplo r150), ? ayuda"));
}

void leerComando() {
  if (!Serial.available()) return;
  String linea = Serial.readStringUntil('\n');
  linea.trim();
  if (linea.length() == 0) return;

  if (linea.charAt(0) == 'r' || linea.charAt(0) == 'R') {
    float valor = linea.substring(1).toFloat();
    if (valor >= 10.0) {
      R_OHM = valor;
      Serial.printf("Resistencia configurada: %.0f ohm\n", R_OHM);
    } else {
      Serial.println(F("Valor no válido. Ejemplo: r150"));
    }
    return;
  }
  mostrarAyuda();
}

void setup() {
  Serial.begin(115200);
  delay(300);
  Serial.println();
  Serial.println(F("=== Simulador 4: medidor de corriente del LED ==="));

  pinMode(PIN_LED, OUTPUT);
  digitalWrite(PIN_LED, HIGH);
  mostrarAyuda();
}

void loop() {
  leerComando();

  if (millis() - ultimoInforme >= 1000) {
    ultimoInforme = millis();

    float vLed = leerTensionA0();
    float vResistencia = V_PIN - vLed;
    float corrienteMa = vResistencia / R_OHM * 1000.0;
    if (corrienteMa < 0) corrienteMa = 0;

    const char* estado;
    if (corrienteMa > LIMITE_MA)  estado = "DEMASIADA CORRIENTE: usá una resistencia mayor";
    else if (corrienteMa < 4.0)   estado = "Brillo bajo";
    else                          estado = "Bien";

    Serial.println(F("------------------------------------------------"));
    Serial.printf("Tensión del pin        : %.2f V (configurada)\n", V_PIN);
    Serial.printf("Tensión del LED (Vf)   : %.2f V (medida en A0)\n", vLed);
    Serial.printf("Tensión en R (%.0f ohm) : %.2f V\n", R_OHM, vResistencia);
    Serial.printf("Corriente              : %.1f mA  -> %s\n", corrienteMa, estado);
  }
}
Valores típicos: la caída de tensión real de cada LED varía entre modelos. Medila con el multímetro para ajustar el cálculo.
